"""
Prediction Module
Purpose: Load model and make predictions
"""

import joblib
import pandas as pd
import numpy as np
from typing import List, Dict, Any, Union
import logging

logger = logging.getLogger(__name__)

class ModelPredictor:
    """
    Loads model and makes predictions
    """
    
    def __init__(self, model_path: str = 'models/model.joblib'):
        """
        Initialize predictor with trained model
        """
        self.model = joblib.load(model_path)
        logger.info(f"Model loaded from {model_path}")
    
    def predict_single(self, features: Union[List, Dict]) -> Union[int, float]:
        """
        Predict for a single sample
        
        Purpose:
        - Make individual predictions
        - Convert dict to list if needed
        """
        if isinstance(features, dict):
            features = list(features.values())
        
        X = np.array(features).reshape(1, -1)
        return self.model.predict(X)[0]
    
    def predict_batch(self, df: pd.DataFrame, feature_cols: List[str]) -> np.ndarray:
        """
        Predict for batch of samples
        
        Purpose:
        - Efficient batch predictions
        - Process multiple samples at once
        """
        X = df[feature_cols]
        return self.model.predict(X)
