"""
Dask Processor
Purpose: Optimize processing for large datasets using Dask
"""

import dask.dataframe as dd
import pandas as pd
from typing import Dict, Any
import logging

logger = logging.getLogger(__name__)

class DaskTransformer:
    """
    Uses Dask for parallel processing of large datasets
    """
    
    def __init__(self, n_workers: int = 4):
        self.n_workers = n_workers
    
    def load_data(self, file_path: str) -> dd.DataFrame:
        """
        Load large dataset using Dask
        
        Purpose:
        - Handle datasets larger than memory
        - Automatically partition data for parallel processing
        """
        logger.info(f"Loading data from {file_path} with Dask")
        return dd.read_csv(file_path, assume_missing=True)
    
    def transform_pipeline(self, df: dd.DataFrame, 
                           operations: list) -> dd.DataFrame:
        """
        Apply multiple transform operations in sequence
        
        Purpose:
        - Chain transformations without materializing intermediate results
        - Lazy evaluation for efficiency
        """
        result = df
        for op in operations:
            op_type = op.get('type')
            op_params = op.get('params', {})
            
            if op_type == 'fillna':
                result = result.fillna(op_params.get('value', 0))
            elif op_type == 'dropna':
                result = result.dropna(subset=op_params.get('subset', None))
            elif op_type == 'cast':
                for col, dtype in op_params.get('columns', {}).items():
                    result[col] = result[col].astype(dtype)
            elif op_type == 'groupby_agg':
                result = result.groupby(op_params.get('by')).agg(op_params.get('agg', {}))
            
            logger.info(f"Applied transform: {op_type}")
        
        return result
    
    def compute_to_pandas(self, df: dd.DataFrame, persist: bool = False) -> pd.DataFrame:
        """
        Compute Dask DataFrame to pandas
        
        Purpose:
        - Materialize the computation result
        - Use persist to cache in memory for repeated access
        """
        logger.info("Computing Dask DataFrame to pandas")
        if persist:
            df = df.persist()
        return df.compute()
